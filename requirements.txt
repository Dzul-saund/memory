# Core runtime dependencies for the Polymarket BTC Up/Down 5m bot.
#
# Only `requests` + `python-dotenv` are required to run the bot in DRY-RUN
# (simulation) mode. `py-clob-client` is only needed to fetch your real
# balance and to place real orders (live mode).
requests>=2.31.0
python-dotenv>=1.0.0
py-clob-client==0.34.6
# Required for live trading on the new Polymarket deposit-wallet accounts
# (signature_type 3 / POLY_1271). The bot's live trader uses this client.
py-clob-client-v2>=1.0.1
# Two DIFFERENT websocket libraries, both required — the names are confusingly
# similar and one does not substitute for the other:
#   websocket-client -> module `websocket`  (sync)  btc_bot/pricefeed.py, bookfeed.py
#   websockets       -> module `websockets` (async) fast_monitor.py, pm_view.py
# fast_monitor is imported by the jump engine, so `websockets` is a hard
# dependency of the bot, not an optional extra. It was missing here and only
# surfaced on a clean server, where nothing had installed it beforehand.
websocket-client>=1.6.0
websockets>=12.0

# Optional accelerators. Every use site falls back cleanly when they are
# absent, so nothing breaks without them — but this bot trades on a ~288ms
# lead, so on a Linux VPS both are worth having.
#   orjson: JSON parsing ~5-10x faster (Rust)
#   uvloop: asyncio event loop 2-4x faster; Linux only, hence the marker,
#           because pip cannot build it on Windows
orjson>=3.9.0
uvloop>=0.19.0; sys_platform != "win32"

# Test runner. It used to arrive only as a transitive dependency of
# poly_eip712_structs, which is luck, not a contract - naming it here keeps
# `pytest tests/` working if that package ever drops it.
pytest>=8.0.0
